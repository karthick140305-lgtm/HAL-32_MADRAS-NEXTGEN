/**
 * PHANTOM-VAULT · Client-Side AES-256-GCM Encryption Module
 * Government-grade client encryption using Web Crypto API (FIPS-140-3 validated in Chrome/Edge).
 * 
 * SECURITY DESIGN:
 * - Encryption occurs entirely client-side before any network transmission
 * - Envelope encryption: per-file DEK wrapped by operator KEK
 * - KEK derived from passphrase via PBKDF2 (never stored)
 * - AES-256-GCM provides authenticated encryption (confidentiality + integrity)
 * - 96-bit random IV per encryption (NIST SP 800-38D compliant)
 */

const CryptoEngine = (() => {
  const ALGORITHM = 'AES-GCM';
  const KEY_LENGTH = 256;            // bits — AES-256
  const IV_LENGTH = 12;              // bytes — 96-bit IV (NIST SP 800-38D §5.2.1.1)
  const TAG_LENGTH = 128;            // bits — maximum GCM tag length
  const SALT_LENGTH = 32;            // bytes — 256-bit KDF salt
  const PBKDF2_ITERATIONS = 600_000; // OWASP 2024 PBKDF2-SHA256 recommendation
  const VERSION = 'PV-1.0';

  /**
   * Generate cryptographically secure random bytes.
   * Uses OS-seeded CSPRNG via Web Crypto — NOT Math.random().
   */
  function randomBytes(length) {
    const buf = new Uint8Array(length);
    crypto.getRandomValues(buf); // CSPRNG — safe for key material
    return buf;
  }

  /**
   * Derive a Key Encryption Key (KEK) from operator passphrase.
   * PBKDF2 with high iteration count provides brute-force resistance.
   * Salt ensures uniqueness across different derivations.
   * 
   * @returns {CryptoKey} — wrapped in browser sandbox, not extractable
   */
  async function deriveKEK(passphrase, salt) {
    const encoder = new TextEncoder();
    const keyMaterial = await crypto.subtle.importKey(
      'raw',
      encoder.encode(passphrase),
      'PBKDF2',
      false,           // non-extractable — cannot be read back as raw bytes
      ['deriveKey']
    );
    return await crypto.subtle.deriveKey(
      {
        name: 'PBKDF2',
        salt,
        iterations: PBKDF2_ITERATIONS,
        hash: 'SHA-256',
      },
      keyMaterial,
      { name: ALGORITHM, length: KEY_LENGTH },
      false,           // KEK itself is non-extractable
      ['wrapKey', 'unwrapKey']
    );
  }

  /**
   * Generate a per-file Data Encryption Key (DEK).
   * Each file gets a unique random 256-bit key.
   * DEK is extractable ONLY for wrapping — never stored raw.
   */
  async function generateDEK() {
    return await crypto.subtle.generateKey(
      { name: ALGORITHM, length: KEY_LENGTH },
      true,            // extractable=true required for wrapKey operation
      ['encrypt', 'decrypt']
    );
  }

  /**
   * Encrypt file contents with AES-256-GCM.
   * 
   * AUTHENTICATED ENCRYPTION:
   * - Ciphertext is encrypted AND authenticated (AEAD)
   * - Any bit flip in ciphertext or AAD causes decryption failure
   * - AAD (Additional Authenticated Data) binds document metadata to ciphertext
   *   preventing ciphertext transplant attacks
   */
  async function encryptFileBuffer(fileBuffer, dek, aadString) {
    const iv = randomBytes(IV_LENGTH); // unique per encryption — CRITICAL
    const aad = new TextEncoder().encode(aadString);
    
    const ciphertext = await crypto.subtle.encrypt(
      {
        name: ALGORITHM,
        iv,
        additionalData: aad,  // authenticated but not encrypted
        tagLength: TAG_LENGTH, // 128-bit auth tag appended to output
      },
      dek,
      fileBuffer
    );
    return { ciphertext, iv };
  }

  /**
   * Main encryption pipeline: file → encrypted envelope.
   * Returns a serializable package safe for transmission to server.
   * 
   * Package schema:
   * {
   *   version:    string  — format version for forward compatibility
   *   salt:       base64  — KDF salt (random, not secret)
   *   kdfIter:    number  — PBKDF2 iterations (for verification)
   *   iv:         base64  — AES-GCM IV (random, not secret)
   *   wrappedDEK: base64  — DEK encrypted under KEK (AES-KW)
   *   ciphertext: base64  — encrypted file content + auth tag
   *   aad:        string  — cleartext metadata bound to ciphertext
   *   docId:      string  — server-side document identifier
   * }
   */
  async function encryptFile(fileBuffer, passphrase, metadata = {}) {
    const salt = randomBytes(SALT_LENGTH);
    const kek = await deriveKEK(passphrase, salt);
    const dek = await generateDEK();
    
    // AAD binds operator, classification, and document ID to ciphertext
    const docId = crypto.randomUUID();
    const aad = `${metadata.operatorId}:${metadata.classification}:${metadata.domain}:${docId}`;
    
    const { ciphertext, iv } = await encryptFileBuffer(fileBuffer, dek, aad);
    
    // Wrap DEK under KEK using AES-KW (RFC 3394)
    const wrappedDEK = await crypto.subtle.wrapKey(
      'raw', dek, kek,
      { name: ALGORITHM, iv: randomBytes(IV_LENGTH), tagLength: TAG_LENGTH }
    );
    
    return {
      version: VERSION,
      docId,
      salt: toBase64(salt),
      kdfIter: PBKDF2_ITERATIONS,
      iv: toBase64(iv),
      wrappedDEK: toBase64(wrappedDEK),
      ciphertext: toBase64(ciphertext),
      aad,
      metadata: {
        classification: metadata.classification,
        domain: metadata.domain,
        filename: metadata.filename,  // filename can be encrypted separately
        uploadTimestamp: Date.now(),
      }
    };
  }

  /**
   * Decrypt an encrypted envelope for authorized users.
   * Requires original passphrase (KEK derivation).
   * Verifies authentication tag before returning any plaintext.
   */
  async function decryptFile(envelope, passphrase) {
    const salt = fromBase64(envelope.salt);
    const kek = await deriveKEK(passphrase, salt);
    
    // Unwrap DEK using KEK
    let dek;
    try {
      const wrappedDEKBuf = fromBase64(envelope.wrappedDEK);
      const wrapIv = wrappedDEKBuf.slice(0, IV_LENGTH);
      dek = await crypto.subtle.unwrapKey(
        'raw', wrappedDEKBuf.slice(IV_LENGTH),
        kek, { name: ALGORITHM, iv: wrapIv, tagLength: TAG_LENGTH },
        { name: ALGORITHM, length: KEY_LENGTH }, false, ['decrypt']
      );
    } catch (e) {
      throw new Error('KEY_UNWRAP_FAILED: Invalid passphrase or corrupted wrapped key');
    }
    
    // Decrypt and verify authentication tag
    try {
      const plaintext = await crypto.subtle.decrypt(
        {
          name: ALGORITHM,
          iv: fromBase64(envelope.iv),
          additionalData: new TextEncoder().encode(envelope.aad),
          tagLength: TAG_LENGTH,
        },
        dek,
        fromBase64(envelope.ciphertext)
      );
      return plaintext; // ArrayBuffer — caller decodes as needed
    } catch (e) {
      throw new Error('AUTH_TAG_VERIFICATION_FAILED: Ciphertext tampered or key mismatch');
    }
  }

  // Base64 helpers for serialization
  function toBase64(buf) {
    return btoa(String.fromCharCode(...new Uint8Array(buf)));
  }
  function fromBase64(b64) {
    return Uint8Array.from(atob(b64), c => c.charCodeAt(0));
  }

  return { encryptFile, decryptFile, randomBytes };
})();

/**
 * PHANTOM-VAULT · Key Management Module
 * Handles key rotation, forward secrecy, and domain key isolation.
 */

const KeyManager = (() => {
  // In-memory key store — never persisted to disk
  const _sessionKeys = new Map();
  let _masterKeyHandle = null;
  let _keyVersion = 0;

  /**
   * Initialize session key via ECDH key agreement.
   * Provides forward secrecy: ephemeral key pair per session.
   * Even if long-term keys are later compromised, past sessions remain secure.
   */
  async function initSessionECDH(serverPublicKeyJWK) {
    // Generate ephemeral client key pair
    const clientKeyPair = await crypto.subtle.generateKey(
      { name: 'ECDH', namedCurve: 'P-384' },  // P-384 for NSA Suite B compliance
      false, ['deriveKey']
    );

    const serverPublicKey = await crypto.subtle.importKey(
      'jwk', serverPublicKeyJWK,
      { name: 'ECDH', namedCurve: 'P-384' },
      false, []
    );

    // Derive shared session key — never transmitted
    const sessionKey = await crypto.subtle.deriveKey(
      { name: 'ECDH', public: serverPublicKey },
      clientKeyPair.privateKey,
      { name: 'AES-GCM', length: 256 },
      false, ['encrypt', 'decrypt']
    );

    const sessionId = crypto.randomUUID();
    _sessionKeys.set(sessionId, {
      key: sessionKey,
      created: Date.now(),
      ttl: 4 * 3600 * 1000,  // 4-hour session TTL
      clientPublicKey: await crypto.subtle.exportKey('jwk', clientKeyPair.publicKey)
    });

    return { sessionId, clientPublicKey: _sessionKeys.get(sessionId).clientPublicKey };
  }

  /**
   * Key rotation: re-wrap all DEKs with new KEK.
   * Old KEK is zeroed. Forward secrecy: past compromises cannot decrypt new ciphertext.
   * This is the ONLY operation that requires temporary access to DEK in memory.
   */
  async function rotateKeys(oldPassphrase, newPassphrase, encryptedDocs) {
    const rotated = [];
    const newSalt = crypto.getRandomValues(new Uint8Array(32));
    _keyVersion++;

    for (const doc of encryptedDocs) {
      // Unwrap DEK with old KEK
      const oldSalt = fromBase64(doc.envelope.salt);
      const oldKEK = await deriveKEK(oldPassphrase, oldSalt);
      const dek = await unwrapDEK(doc.envelope.wrappedDEK, oldKEK);

      // Re-wrap DEK with new KEK — ciphertext unchanged
      const newKEK = await deriveKEK(newPassphrase, newSalt);
      const newWrappedDEK = await rewrapDEK(dek, newKEK);

      rotated.push({
        ...doc,
        envelope: {
          ...doc.envelope,
          salt: toBase64(newSalt),
          wrappedDEK: newWrappedDEK,
          keyVersion: _keyVersion,
        }
      });
    }

GCM_IV_BYTES     = 12    # 96-bit IV (NIST SP 800-38D)
GCM_TAG_BYTES    = 16    # 128-bit authentication tag
PBKDF2_ITERS     = 600_000  # OWASP 2024 minimum for SHA-256
SALT_BYTES       = 32    # 256-bit KDF salt
HMAC_KEY_BYTES   = 64    # 512-bit SSE trapdoor key


class SecureUploadProcessor:
    """
    Receives client-side-encrypted documents.
    Validates envelope, extracts HMAC search tokens, stores ciphertext only.
    
    THREAT MODEL: This server may be fully compromised.
    No operation here should require or expose plaintext.
    """

    def __init__(self, hsm_interface, storage_backend, hmac_key: bytes):
        self.hsm = hsm_interface      # HSM abstraction (PKCS#11 in prod)
        self.storage = storage_backend
        # SSE trapdoor key — loaded from HSM, never stored in DB
        self._trapdoor_key = hmac_key

    def process_upload(self, encrypted_envelope: dict, operator_id: str) -> dict:
        """
        Process an encrypted document upload.
        
        INPUT: Client-side encrypted envelope (server sees no plaintext)
        OUTPUT: Storage manifest with document ID and HMAC index tokens
        """
        # 1. Validate envelope structure
        self._validate_envelope(encrypted_envelope)

        # 2. Compute document fingerprint for integrity
        ciphertext_bytes = base64.b64decode(encrypted_envelope['ciphertext'])
        doc_fingerprint = hashlib.sha3_256(ciphertext_bytes).hexdigest()

        # 3. Server does NOT decrypt — validates auth metadata only
        doc_id = encrypted_envelope.get('docId') or secrets.token_hex(16)

        # 4. Build storage record — only encrypted data
        storage_record = {
            'doc_id':          doc_id,
            'ciphertext':      encrypted_envelope['ciphertext'],  # opaque bytes
            'iv':              encrypted_envelope['iv'],
            'wrapped_dek':     encrypted_envelope['wrappedDEK'],
            'salt':            encrypted_envelope['salt'],
            'kdf_iterations':  encrypted_envelope.get('kdfIter', PBKDF2_ITERS),
            'aad':             encrypted_envelope.get('aad', ''),
            'version':         encrypted_envelope.get('version'),
            'fingerprint':     doc_fingerprint,
            'operator_id':     operator_id,
            'upload_ts':       datetime.now(timezone.utc).isoformat(),
            'classification':  encrypted_envelope.get(
                                   'metadata', {}).get('classification', 'UNKNOWN'),
        }

        # 5. Persist encrypted record
        self.storage.store_encrypted(storage_record)

        return {
            'doc_id':      doc_id,
            'fingerprint': doc_fingerprint,
            'status':      'STORED_ENCRYPTED',
        }

    def _validate_envelope(self, envelope: dict):
        required = {'ciphertext', 'iv', 'wrappedDEK', 'salt', 'version'}
        missing = required - set(envelope.keys())
        if missing:
            raise ValueError(f"Invalid envelope — missing fields: {missing}")
        iv_bytes = base64.b64decode(envelope['iv'])
        if len(iv_bytes) != GCM_IV_BYTES:
            raise ValueError("Invalid IV length — must be 96 bits")


class ServerSideEncryptionFallback:
    """
    FALLBACK: If client cannot perform encryption (non-browser env).
    Performs AES-256-GCM encryption server-side IN MEMORY.
    File never written to disk in plaintext.
    Key material sourced from HSM — never plaintext in DB.
    """

    def encrypt_in_memory(self, plaintext_bytes: bytes,
                            hsm_key_id: str) -> dict:
        # Generate per-file DEK from HSM-stored master key
        dek = secrets.token_bytes(AES_KEY_BYTES)  # In prod: HSM.generate_dek()
        iv  = secrets.token_bytes(GCM_IV_BYTES)   # Must be unique per key use

        # AES-256-GCM authenticated encryption
        aesgcm = AESGCM(dek)
        ciphertext_with_tag = aesgcm.encrypt(iv, plaintext_bytes, None)

        # Envelope: wrap DEK under HSM master key
        wrapped_dek = self.hsm.wrap_key(dek, hsm_key_id)

        # Zeroize DEK from memory immediately
        dek = b'\x00' * AES_KEY_BYTES  # NOTE: Python GC may not honor this immediately
        del dek                          # Production: use ctypes for guaranteed zeroization

        return {
            'iv':         base64.b64encode(iv).decode(),
            'ciphertext': base64.b64encode(ciphertext_with_tag).decode(),
            'wrapped_dek': base64.b64encode(wrapped_dek).decode(),
            'hsm_key_id': hsm_key_id,
        }

    def decrypt_for_authorized(self, encrypted_record: dict,
                               operator_clearance: str) -> bytes:
        # Authorization check BEFORE any crypto operation
        if not self._check_clearance(operator_clearance,
                                      encrypted_record['classification']):
            raise PermissionError("INSUFFICIENT_CLEARANCE")

        # Unwrap DEK via HSM
        wrapped_dek = base64.b64decode(encrypted_record['wrapped_dek'])
        dek = self.hsm.unwrap_key(wrapped_dek, encrypted_record['hsm_key_id'])

        # Decrypt and verify authentication tag
        iv         = base64.b64decode(encrypted_record['iv'])
        ciphertext = base64.b64decode(encrypted_record['ciphertext'])
        aesgcm     = AESGCM(dek)

        try:
            plaintext = aesgcm.decrypt(iv, ciphertext, None)
        except Exception:
            raise ValueError("AUTH_TAG_VERIFICATION_FAILED")
        finally:
            dek = b'\x00' * len(dek)  # Zeroize DEK after use

        return plaintext

/**
 * Secure Upload Handler
 * Receives client-encrypted envelopes. Never decrypts them.
 * Validates structure, stores opaque ciphertext, generates HMAC search tokens.
 */
class SecureUploadController {
  constructor(storageAdapter, hsmClient, auditLogger) {
    this.storage = storageAdapter;
    this.hsm     = hsmClient;
    this.audit   = auditLogger;
  }

  /**
   * POST /api/upload
   * Expects: multipart/form-data with 'envelope' JSON field + optional binary
   */
  handleUpload = async (req, res) => {
    const operatorId = req.user.badgeId;       // from JWT/PKI auth middleware
    const sessionKey  = req.sessionKey;         // from mTLS session

    try {
      const envelope = JSON.parse(req.body.envelope);
      this._validateEnvelope(envelope);

      const ciphertextBuf = Buffer.from(envelope.ciphertext, 'base64');

      // Fingerprint ciphertext for integrity verification (not a MAC)
      const fingerprint = crypto
        .createHash('sha3-256')
        .update(ciphertextBuf)
        .digest('hex');

      // Generate SSE HMAC tokens from metadata (not from content)
      const hmacTokens = this._generateSearchTokens(envelope, operatorId);

      const record = {
        docId:         envelope.docId || crypto.randomUUID(),
        ciphertext:    envelope.ciphertext,   // stays encrypted
        iv:            envelope.iv,
        wrappedDek:    envelope.wrappedDEK,
        salt:          envelope.salt,
        kdfIterations: envelope.kdfIter,
        aad:           envelope.aad,
        fingerprint,
        hmacTokens,                           // opaque search index
        operatorId,
        classification: envelope.metadata?.classification,
        uploadedAt:    new Date().toISOString(),
      };

      await this.storage.save(record);

      await this.audit.log({
        event:       'DOCUMENT_UPLOADED',
        docId:       record.docId,
        operatorId,
        fingerprint,
        timestamp:   record.uploadedAt,
      });

      res.json({ docId: record.docId, fingerprint, status: 'STORED_ENCRYPTED' });

    } catch (err) {
      this.audit.error({ event: 'UPLOAD_FAILED', operatorId, error: err.message });
      res.status(400).json({ error: err.message });
    }
  };

  /**
   * Generate HMAC-SHA3-512 search tokens from document metadata.
   * These are the SSE index tokens — server cannot invert them.
   */
  _generateSearchTokens(envelope, operatorId) {
    const trapdoorKey = this.hsm.getTrapdoorKey(); // from HSM, not DB
    const metadata    = envelope.metadata || {};
    const fields      = [
      metadata.classification,
      metadata.domain,
      metadata.filename?.replace(/\.[^.]+$/, ''),
    ].filter(Boolean);

    return fields.map(f =>
      crypto.createHmac('sha3-512', trapdoorKey)
             .update(f.toLowerCase().trim())
             .digest('hex')
    );
  }

  _validateEnvelope(env) {
    const required = ['ciphertext', 'iv', 'wrappedDEK', 'salt', 'version'];
    for (const field of required) {
      if (!env[field]) throw new Error(`Missing envelope field: ${field}`);
    }
    const iv = Buffer.from(env.iv, 'base64');
    if (iv.length !== CONSTANTS.GCM_IV_BYTES) {
      throw new Error('Invalid IV — must be 96 bits (12 bytes)');
    }
  }
}

/**
 * Server-side AES-256-GCM (fallback for non-browser clients).
 * Key sourced from HSM — never from DB or environment variables.
 */
class ServerCrypto {
  encryptBuffer(plaintext, key) {
    // IV must be cryptographically random and unique per key use
    const iv     = crypto.randomBytes(CONSTANTS.GCM_IV_BYTES);
    const cipher = crypto.createCipheriv('aes-256-gcm', key, iv);

    const encrypted  = Buffer.concat([cipher.update(plaintext), cipher.final()]);
    const authTag    = cipher.getAuthTag(); // 16 bytes = 128-bit GCM tag

    // Pack: IV ‖ authTag ‖ ciphertext — all components needed for decryption
    return { iv, authTag, ciphertext: encrypted };
  }

  decryptBuffer(encRecord, key) {
    const decipher = crypto.createDecipheriv(
      'aes-256-gcm', key, encRecord.iv
    );
    // setAuthTag MUST be called before finalize — this is where tampering is detected
    decipher.setAuthTag(encRecord.authTag);

    try {
      const decrypted = Buffer.concat([
        decipher.update(encRecord.ciphertext),
        decipher.final()  // throws if auth tag fails
      ]);
      return decrypted;
    } catch (e) {
      throw new Error('AUTH_TAG_VERIFICATION_FAILED: Data integrity compromised');
    }
  }
}
